## 01. initial schema
==============================================================================
-- Initial schema migration matching the existing Drizzle schema
-- This creates all core tables for the Hyperfocus application

CREATE TABLE periods_table (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    type TEXT NOT NULL CHECK (type IN ('session', 'day', 'week', 'month')),
    started BOOLEAN NOT NULL DEFAULT 0,
    finished BOOLEAN NOT NULL DEFAULT 0,
    started_at INTEGER,
    finished_at INTEGER,
    created_at INTEGER NOT NULL DEFAULT (unixepoch() * 1000),
    parent_id TEXT REFERENCES periods_table(id),
    duration INTEGER NOT NULL DEFAULT 0 CHECK (duration >= 0),
    focused_time REAL,
    archived BOOLEAN DEFAULT 0,
    position INTEGER NOT NULL DEFAULT 0 CHECK (position >= 0 AND position <= 999),
    repeat_id TEXT REFERENCES repeats_table(id),
    plan_reality_rating INTEGER,
    CHECK (NOT (started = 0 AND finished = 1))
);

CREATE TABLE tasks_table (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    completed BOOLEAN NOT NULL DEFAULT 0,
    subtasks TEXT NOT NULL DEFAULT '[]', -- JSON as text
    period_id TEXT NOT NULL REFERENCES periods_table(id),
    position INTEGER NOT NULL DEFAULT 0,
    migrated_from_task_id TEXT REFERENCES tasks_table(id)
);

CREATE TABLE checkins_table (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pmf TEXT CHECK (pmf IN ('very_disappointed', 'somewhat_disappointed', 'not_disappointed')),
    created_at INTEGER NOT NULL DEFAULT (unixepoch() * 1000)
);

CREATE TABLE repeats_table (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    duration INTEGER NOT NULL CHECK (duration >= 0),
    position INTEGER NOT NULL DEFAULT 0 CHECK (position >= 0 AND position <= 999),
    archived BOOLEAN NOT NULL DEFAULT 0
);

-- Essential indexes for performance
CREATE INDEX idx_tasks_period ON tasks_table(period_id);
CREATE INDEX idx_periods_parent ON periods_table(parent_id);
CREATE INDEX idx_periods_type ON periods_table(type);
CREATE INDEX idx_periods_archived ON periods_table(archived);

## 25. collapse later periods
==============================================================================
-- Collapse three legacy Later buckets (later-month, later-week, later-day)
-- into a single unified `later` period.

-- 1. Insert the unified Later period (idempotent).
INSERT OR IGNORE INTO periods_table (
    id, title, type, started, finished,
    started_at, finished_at, created_at,
    parent_id, duration, focused_time,
    archived, position, repeat_id, active
) VALUES (
    'later', 'Later', 'month',
    0, 0,
    NULL, NULL, 0,
    NULL, 0, NULL,
    0, 0, NULL, 0
);

-- 2. Move tasks from legacy Later periods into `later`.
--    Merge order: later-month tasks first, then later-week, then later-day.
--    Uses a window function to assign contiguous positions.
UPDATE tasks_table
SET period_id = 'later',
    position = (
        SELECT new_pos FROM (
            SELECT t2.id,
                   ROW_NUMBER() OVER (
                       ORDER BY
                           CASE t2.period_id
                               WHEN 'later-month' THEN 0
                               WHEN 'later-week'  THEN 1
                               WHEN 'later-day'   THEN 2
                           END,
                           t2.position ASC,
                           t2.rowid ASC
                   ) - 1 AS new_pos
            FROM tasks_table t2
            WHERE t2.period_id IN ('later-month', 'later-week', 'later-day')
        ) AS ranked
        WHERE ranked.id = tasks_table.id
    )
WHERE period_id IN ('later-month', 'later-week', 'later-day');

-- 3. Delete the legacy Later period rows.
DELETE FROM periods_table
WHERE id IN ('later-month', 'later-week', 'later-day');

CREATE TRIGGER reject_task_root_color_key_non_long_term_insert
BEFORE INSERT ON tasks_table
FOR EACH ROW
WHEN NEW.root_color_key IS NOT NULL
    AND NOT EXISTS (
        SELECT 1
        FROM cycles_table
        WHERE id = NEW.cycle_id
          AND type = 'month'
    )
BEGIN
    SELECT RAISE(ABORT, 'root_color_key requires a Long-term cycle');
END;

CREATE TRIGGER reject_task_root_color_key_non_long_term_update
BEFORE UPDATE OF cycle_id, root_color_key ON tasks_table
FOR EACH ROW
WHEN NEW.root_color_key IS NOT NULL
    AND NOT EXISTS (
        SELECT 1
        FROM cycles_table
        WHERE id = NEW.cycle_id
          AND type = 'month'
    )
BEGIN
    SELECT RAISE(ABORT, 'root_color_key requires a Long-term cycle');
END;

CREATE TRIGGER reject_cycle_type_change_with_root_colors
BEFORE UPDATE OF type ON cycles_table
FOR EACH ROW
WHEN NEW.type != 'month'
    AND EXISTS (
        SELECT 1
        FROM tasks_table
        WHERE cycle_id = NEW.id
          AND root_color_key IS NOT NULL
    )
BEGIN
    SELECT RAISE(ABORT, 'cycles with root colors must stay Long-term');
END;
